cmake_minimum_required(VERSION 3.22)
project(plezy_torrent LANGUAGES C CXX)
include(FetchContent)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 OLD)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(PLEZY_TORRENT_SYSTEM "Use a system libtorrent (development only)" OFF)
if(PLEZY_TORRENT_SYSTEM)
  find_package(LibtorrentRasterbar 2.0 CONFIG REQUIRED)
else()
  # These hashes also pin the source archives included in release notices.
  FetchContent_Declare(plezy_boost
    URL https://archives.boost.io/release/1.86.0/source/boost_1_86_0.tar.bz2
    URL_HASH SHA256=1bed88e40401b2cb7a1f76d4bab499e352fa4d0c5f31c0dbae64e24d34d7513b)
  FetchContent_GetProperties(plezy_boost)
  if(NOT plezy_boost_POPULATED)
    FetchContent_Populate(plezy_boost)
  endif()
  set(BOOST_ROOT "${plezy_boost_SOURCE_DIR}")
  # Explicit headers bypass cross-toolchain sysroot rewriting in FindBoost.
  set(Boost_INCLUDE_DIR "${plezy_boost_SOURCE_DIR}")
  set(Boost_NO_SYSTEM_PATHS ON)
  set(Boost_NO_BOOST_CMAKE ON)
  # Keep these scoped to this plugin; never change other Flutter plugins'
  # shared-library or test settings in the parent project's CMake cache.
  set(BUILD_SHARED_LIBS OFF)
  set(build_tests OFF)
  set(build_examples OFF)
  set(build_tools OFF)
  set(logging OFF)
  set(i2p OFF)
  # DHT/PEX and HTTP/UDP trackers, with libtorrent's built-in crypto. Never
  # accidentally link host OpenSSL into a mobile/TV cross build.
  set(CMAKE_DISABLE_FIND_PACKAGE_OpenSSL TRUE)
  set(CMAKE_DISABLE_FIND_PACKAGE_GnuTLS TRUE)
  set(CMAKE_DISABLE_FIND_PACKAGE_LibGcrypt TRUE)
  FetchContent_Declare(libtorrent
    URL https://github.com/arvidn/libtorrent/releases/download/v2.0.11/libtorrent-rasterbar-2.0.11.tar.gz
    URL_HASH SHA256=f0db58580f4f29ade6cc40fa4ba80e2c9a70c90265cd77332d3cdec37ecf1e6d)
  FetchContent_MakeAvailable(libtorrent)
  add_library(LibtorrentRasterbar::torrent-rasterbar ALIAS torrent-rasterbar)
  install(FILES "${libtorrent_SOURCE_DIR}/COPYING" DESTINATION share/plezy/licenses RENAME libtorrent.txt)
  install(FILES "${plezy_boost_SOURCE_DIR}/LICENSE_1_0.txt" DESTINATION share/plezy/licenses RENAME boost.txt)
endif()
add_library(plezy_torrent SHARED plezy_torrent.cpp)
target_compile_features(plezy_torrent PRIVATE cxx_std_17)
target_link_libraries(plezy_torrent PRIVATE LibtorrentRasterbar::torrent-rasterbar)
set_target_properties(plezy_torrent PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(ANDROID)
  target_link_options(plezy_torrent PRIVATE "-Wl,-z,max-page-size=16384")
endif()
if(APPLE)
  set_target_properties(plezy_torrent PROPERTIES FRAMEWORK TRUE FRAMEWORK_VERSION A
    MACOSX_FRAMEWORK_IDENTIFIER org.plezy.torrent OUTPUT_NAME PlezyTorrent
    PUBLIC_HEADER plezy_torrent.h)
endif()
option(PLEZY_TORRENT_TESTS "Build native loopback transfer tests" OFF)
if(PLEZY_TORRENT_TESTS)
  enable_testing()
  target_compile_definitions(plezy_torrent PRIVATE PLEZY_TORRENT_TEST_LOOPBACK=1)
  add_executable(torrent_test ../test/torrent_test.cpp)
  target_compile_features(torrent_test PRIVATE cxx_std_17)
  target_link_libraries(torrent_test PRIVATE plezy_torrent LibtorrentRasterbar::torrent-rasterbar)
  add_test(NAME torrent_verified_range COMMAND torrent_test)
endif()
