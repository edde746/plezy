cmake_minimum_required(VERSION 3.22)
project(plezy_steam LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

get_filename_component(PLEZY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(PLEZY_STEAM_SOURCE "${PLEZY_ROOT}/.dart_tool/games/steam" CACHE PATH "Prepared, pinned engine sources")
if(NOT EXISTS "${PLEZY_STEAM_SOURCE}/.plezy-recipe")
  message(FATAL_ERROR "Run python3 scripts/games/prepare_steam.py before configuring")
endif()
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PLEZY_STEAM_SOURCE}/mbedtls" "${CMAKE_CURRENT_BINARY_DIR}/mbedtls" EXCLUDE_FROM_ALL)

add_library(plezy_protobuf_c STATIC "${PLEZY_STEAM_SOURCE}/protobuf-c/protobuf-c/protobuf-c.c")
target_include_directories(plezy_protobuf_c PUBLIC "${PLEZY_STEAM_SOURCE}/protobuf-c")

set(PYROWAVE_ROOT "${PLEZY_STEAM_SOURCE}/pyrowave")
if(NOT EXISTS "${PYROWAVE_ROOT}/pyrowave.h")
  message(FATAL_ERROR "Run python3 scripts/games/prepare_steam.py to fetch the pinned Pyrowave source")
endif()
set(GRANITE_VULKAN_SHADER_MANAGER_RUNTIME_COMPILER OFF CACHE BOOL "" FORCE)
set(GRANITE_SHADER_COMPILER_OPTIMIZE OFF CACHE BOOL "" FORCE)
set(GRANITE_TOOLS OFF CACHE BOOL "" FORCE)
set(GRANITE_POSITION_INDEPENDENT ON CACHE BOOL "" FORCE)
set(GRANITE_SHIPPING ON CACHE BOOL "" FORCE)
set(GRANITE_VULKAN_SPIRV_CROSS OFF CACHE BOOL "" FORCE)
set(GRANITE_VULKAN_SYSTEM_HANDLES OFF CACHE BOOL "" FORCE)
set(GRANITE_RENDERER OFF CACHE BOOL "" FORCE)
set(GRANITE_VULKAN_FOSSILIZE OFF CACHE BOOL "" FORCE)
set(GRANITE_PLATFORM "headless" CACHE STRING "" FORCE)
add_subdirectory("${PYROWAVE_ROOT}/Granite" "${CMAKE_CURRENT_BINARY_DIR}/granite" EXCLUDE_FROM_ALL)
add_subdirectory("${PYROWAVE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/pyrowave" EXCLUDE_FROM_ALL)
add_library(plezy_pyrowave_capi STATIC
  "${PYROWAVE_ROOT}/pyrowave_c.cpp"
  "${PYROWAVE_ROOT}/Granite/video/scaler.cpp")
target_include_directories(plezy_pyrowave_capi PUBLIC "${PYROWAVE_ROOT}"
  PRIVATE "${PYROWAVE_ROOT}/shaders" "${PYROWAVE_ROOT}/Granite/video")
target_compile_definitions(plezy_pyrowave_capi PRIVATE PYROWAVE_EXPORT_SYMBOLS)
target_link_libraries(plezy_pyrowave_capi PRIVATE pyrowave granite-vulkan granite-math)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PLEZY_STEAM_SOURCE}/opus" "${CMAKE_CURRENT_BINARY_DIR}/opus" EXCLUDE_FROM_ALL)

set(IHS_ROOT "${PLEZY_STEAM_SOURCE}/ihslib")
file(GLOB_RECURSE IHS_SOURCES CONFIGURE_DEPENDS "${IHS_ROOT}/src/*.c")
list(FILTER IHS_SOURCES EXCLUDE REGEX "/platforms/|/hid/sdl/|/crypto/")
if(NOT UNIX)
  message(FATAL_ERROR "This native target needs a POSIX socket backend")
endif()
add_library(plezy_ihs STATIC ${IHS_SOURCES}
  "${IHS_ROOT}/src/crypto/impl_mbedtls.c"
  "${IHS_ROOT}/src/platforms/ihs_ip_posix.c"
  "${IHS_ROOT}/src/platforms/ihs_udp_posix.c"
  ihs_thread_posix.c)
target_include_directories(plezy_ihs SYSTEM PUBLIC "${IHS_ROOT}/include")
target_include_directories(plezy_ihs PRIVATE "${IHS_ROOT}/src")
target_compile_definitions(plezy_ihs PRIVATE _POSIX_C_SOURCE=200809L _DEFAULT_SOURCE)
find_package(Threads REQUIRED)
target_link_libraries(plezy_ihs PUBLIC plezy_protobuf_c mbedcrypto Threads::Threads m)

add_library(plezy_steam_runtime STATIC steam_runtime.cpp steam_controller.cpp steam_pyrowave.cpp steam_audio.cpp)
target_include_directories(plezy_steam_runtime BEFORE PRIVATE
  "${PYROWAVE_ROOT}/Granite/third_party/khronos/vulkan-headers/include")
target_include_directories(plezy_steam_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${IHS_ROOT}/src")
target_link_libraries(plezy_steam_runtime PUBLIC plezy_ihs opus PRIVATE plezy_pyrowave_capi)
target_compile_options(plezy_steam_runtime PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
  add_library(plezy_steam SHARED steam_jni.cpp)
  target_link_libraries(plezy_steam PRIVATE plezy_steam_runtime log)
endif()

option(PLEZY_STEAM_TESTS "Build transport and adapter regression tests" OFF)
option(PLEZY_STEAM_GSTREAMER "Build the dedicated Linux game decoder" OFF)
if(PLEZY_STEAM_GSTREAMER)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(STEAM_GST REQUIRED IMPORTED_TARGET gstreamer-app-1.0>=1.20 gstreamer-video-1.0>=1.20)
  add_library(plezy_steam_gstreamer STATIC steam_gstreamer.cpp)
  target_link_libraries(plezy_steam_gstreamer PUBLIC plezy_steam_runtime PkgConfig::STEAM_GST)
  target_compile_options(plezy_steam_gstreamer PRIVATE -Wall -Wextra -Werror)
endif()
if(PLEZY_STEAM_TESTS AND NOT ANDROID)
  enable_testing()
  add_library(ihslib ALIAS plezy_ihs)
  set(TEST_INCLUDES "${IHS_ROOT}/include" "${IHS_ROOT}/src" "${IHS_ROOT}/tests")
  function(ihs_add_test NAME SOURCE)
    add_executable(ihstest_${NAME} ${SOURCE})
    target_include_directories(ihstest_${NAME} PRIVATE ${TEST_INCLUDES})
    target_link_libraries(ihstest_${NAME} PRIVATE plezy_ihs)
    target_compile_options(ihstest_${NAME} PRIVATE -UNDEBUG)
    add_test(NAME ${NAME} COMMAND ihstest_${NAME})
    set_tests_properties(${NAME} PROPERTIES TIMEOUT 45 SKIP_RETURN_CODE 127)
    set(IHSTEST_TARGET ihstest_${NAME} PARENT_SCOPE)
  endfunction()
  add_subdirectory("${IHS_ROOT}/tests/common" tests/common)
  target_link_libraries(ihs-test-session PUBLIC plezy_ihs)
  foreach(name arraylist enumeration crypto authorization udp_broadcast hid_report_batch hid_report_chain hid_report_replace)
    ihs_add_test(${name} "${IHS_ROOT}/tests/test_${name}.c")
  endforeach()
  foreach(name report_data hid_manager)
    ihs_add_test(${name} "${IHS_ROOT}/tests/hid/${name}.c")
    target_link_libraries(${IHSTEST_TARGET} PRIVATE ihs-test-session)
  endforeach()
  ihs_add_test(hid_concurrent_devices "${IHS_ROOT}/tests/hid/test_concurrent_devices.c")
  target_link_libraries(${IHSTEST_TARGET} PRIVATE ihs-test-session)
  add_subdirectory("${IHS_ROOT}/tests/session" tests/session)
  add_executable(plezy_runtime_test tests/runtime_test.cpp)
  target_include_directories(plezy_runtime_test PRIVATE "${IHS_ROOT}/src")
  target_link_libraries(plezy_runtime_test PRIVATE plezy_steam_runtime)
  target_compile_options(plezy_runtime_test PRIVATE -UNDEBUG)
  add_test(NAME plezy_runtime COMMAND plezy_runtime_test)
  set_tests_properties(plezy_runtime PROPERTIES TIMEOUT 15)
  add_executable(plezy_pyrowave_test tests/pyrowave_test.cpp)
  target_link_libraries(plezy_pyrowave_test PRIVATE plezy_steam_runtime)
  target_compile_options(plezy_pyrowave_test PRIVATE -UNDEBUG)
  add_test(NAME plezy_pyrowave_bitstream COMMAND plezy_pyrowave_test)
  set_tests_properties(plezy_pyrowave_bitstream PROPERTIES TIMEOUT 15)
  add_executable(plezy_audio_test tests/audio_test.cpp)
  target_link_libraries(plezy_audio_test PRIVATE plezy_steam_runtime)
  target_compile_options(plezy_audio_test PRIVATE -UNDEBUG)
  add_test(NAME plezy_audio COMMAND plezy_audio_test)
  set_tests_properties(plezy_audio PROPERTIES TIMEOUT 15)
  if(PLEZY_STEAM_GSTREAMER)
    add_executable(plezy_gstreamer_test tests/gstreamer_test.cpp)
    target_link_libraries(plezy_gstreamer_test PRIVATE plezy_steam_gstreamer)
    target_compile_options(plezy_gstreamer_test PRIVATE -UNDEBUG)
    add_test(NAME plezy_gstreamer COMMAND plezy_gstreamer_test)
    set_tests_properties(plezy_gstreamer PROPERTIES TIMEOUT 45)
  endif()
endif()
